# DEF-02 · formatPlantTime() depende de la zona horaria del proceso
# Comando (desde frontend/src/lib, Node 22): node --input-type=module -e "process.env.TZ='<zona>'; const m = await import('./format.ts'); ..."
# Entrada: '2026-09-12T14:00:00Z' (hora de planta de inicio del cambio de M-109)

TZ=UTC             formatPlantTime = 12/09 14:00 (esperado 12/09 14:00) | formatPlantDay = 12/09
TZ=America/Bogota  formatPlantTime = 12/09 09:00 (esperado 12/09 14:00) | formatPlantDay = 12/09
TZ=Asia/Tokyo      formatPlantTime = 12/09 23:00 (esperado 12/09 14:00) | formatPlantDay = 12/09
